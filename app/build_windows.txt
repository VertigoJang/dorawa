# Dorawa 윈도우 실행본(.exe) 빌드 안내
# (윈도우 PC에서, 파이썬 설치된 상태)

python -m venv .venv
.venv\Scripts\activate
pip install -r requirements.txt pyinstaller
pyinstaller --clean --noconfirm dorawa.spec

# 결과: dist\Dorawa.exe  (더블클릭 실행)
